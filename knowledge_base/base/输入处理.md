# 输入处理

Godot 4.x 的输入系统通过 Input 单例和 InputEvent 类处理所有用户输入。

## 1. Input 单例

```gdscript
# 轮询输入（在 _process 或 _physics_process 中使用）
func _process(delta: float) -> void:
    if Input.is_action_pressed("ui_accept"):
        confirm()
    if Input.is_action_just_pressed("jump"):
        jump()

    # 获取输入向量
    var direction := Input.get_vector("ui_left", "ui_right", "ui_up", "ui_down")
    velocity = direction * speed
```

## 2. 输入映射（Input Map）

在项目设置 > 输入映射中定义动作：
- `ui_accept` — 确认（Enter/Space）
- `ui_cancel` — 取消（Escape）
- `ui_left/right/up/down` — 方向键
- 自定义：`jump`、`attack`、`pause` 等

```gdscript
# 代码中检查
if Input.is_action_just_pressed("attack"):
    attack()
```

## 3. _input 事件处理

```gdscript
# 全局输入事件处理
func _input(event: InputEvent) -> void:
    if event.is_action_pressed("ui_cancel"):
        show_pause_menu()
        get_tree().root.set_input_as_handled()

    if event is InputEventMouseButton:
        if event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
            handle_click(event.position)

    if event is InputEventKey:
        if event.keycode == KEY_SPACE:
            pause_toggle()
```

## 4. _unhandled_input 与 _gui_input

```gdscript
# Control 节点的 GUI 输入
func _gui_input(event: InputEvent) -> void:
    if event is InputEventMouseButton:
        if event.button_index == MOUSE_BUTTON_LEFT:
            accept_event()  # 阻止冒泡

# 未被其他节点处理的输入
func _unhandled_input(event: InputEvent) -> void:
    if event.is_action_pressed("shortcut"):
        execute_shortcut()
```

## 5. 鼠标事件

```gdscript
func _input(event: InputEvent) -> void:
    if event is InputEventMouseMotion:
        mouse_position = event.position

    if event is InputEventMouseButton:
        match event.button_index:
            MOUSE_BUTTON_LEFT:
                if event.pressed:
                    start_drag()
                else:
                    end_drag()
            MOUSE_BUTTON_RIGHT:
                cancel_action()
            MOUSE_BUTTON_WHEEL_UP:
                zoom_in()
            MOUSE_BUTTON_WHEEL_DOWN:
                zoom_out()
```

## 6. 触控与手势

```gdscript
func _input(event: InputEvent) -> void:
    if event is InputEventScreenTouch:
        if event.pressed:
            touch_start(event.position)

    if event is InputEventScreenDrag:
        update_drag(event.position, event.relative)
```

## 7. 导航与焦点

```gdscript
# UI 导航
button.grab_focus()                    # 获取焦点
```
